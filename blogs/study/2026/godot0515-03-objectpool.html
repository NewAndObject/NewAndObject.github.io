# 对象池：告别 GC 抖动

# 问题：频繁创建销毁 = 性能杀手

一个射击游戏里：

void FireBullet() {
    var bullet = new Bullet();               // 分配内存
    bullets.Add(bullet);
}

void OnBulletExpired(Bullet b) {
    bullets.Remove(b);                       // 变成垃圾
}

每颗子弹 = 一次堆分配 + 一次 GC 回收。一梭子 30 发，60fps 下每秒成百上千次分配。后果：

  • GC 尖刺：C#/Java/Lua 等语言，GC 触发时游戏卡顿几十到几百毫秒
  • 内存碎片：频繁的小对象分配导致堆碎片化
  • CPU 浪费：new 操作本身比复用慢一个数量级

提前分配一批对象，用的时候拿，用完归还，不销毁、不重新分配。

# 基本实现

class ObjectPool<T> where T : class {
    private Stack<T> pool = new();
    private Func<T> factory;

    public ObjectPool(Func<T> factory, int initialSize) {
        this.factory = factory;
        for (int i = 0; i < initialSize; i++) {
            pool.Push(factory());
        }
    }

    public T Get() {
        return pool.Count > 0 ? pool.Pop() : factory();
    }

    public void Return(T obj) {
        pool.Push(obj);
    }
}

# 子弹系统应用

class BulletPool : ObjectPool<Bullet> {
    public BulletPool() : base(() => new Bullet(), 50) { }

    public Bullet SpawnBullet(Vector3 pos, Vector3 dir) {
        var b = Get();
        b.Reset(pos, dir);       // 重置状态
        b.gameObject.SetActive(true);
        return b;
    }

    public void DespawnBullet(Bullet b) {
        b.gameObject.SetActive(false);
        Return(b);
    }
}

# 在 Godot 中的实现

class_name ObjectPool

var pool: Array[Node] = []
var scene: PackedScene

func _init(scene_path: String, prealloc: int):
    scene = load(scene_path)
    for i in prealloc:
        var obj = scene.instantiate()
        obj.add_to_group("pooled")
        add_child(obj)
        obj.hide()
        pool.append(obj)

func get_object() -> Node:
    if pool.is_empty():
        var obj = scene.instantiate()
        obj.add_to_group("pooled")
        add_child(obj)
        return obj
    var obj = pool.pop_back()
    obj.show()
    return obj

func return_object(obj: Node):
    obj.hide()
    pool.append(obj)

# 比基本实现更强的变体

# 1. 自动回收

class AutoReturnBullet : Bullet {
    public ObjectPool<Bullet> pool;
    float lifetime;

    void Update() {
        lifetime -= Time.deltaTime;
        if (lifetime <= 0) pool.Return(this);
    }
}

# 2. Warming（预热）

关卡加载时预先模拟几帧，让池中对象完成初始化（比如粒子系统的预热）：

void Warmup(int count) {
    var temp = new List<Bullet>();
    for (int i = 0; i < count; i++) temp.Add(Get());
    for (int i = 0; i < count; i++) Return(temp[i]);
}

# 3. 多态池 / 异构池

一个池子里不只有一种对象：

class EffectPool {
    // 按类型分桶
    private Dictionary<string, ObjectPool<Effect>> pools = new();

    public Effect Spawn(string type, Vector3 pos) {
        if (!pools.ContainsKey(type))
            pools[type] = new ObjectPool<Effect>(() => CreateEffect(type), 10);
        var e = pools[type].Get();
        e.Reset(pos);
        return e;
    }
}

# 《使命召唤》的子弹池

IW 引擎的子弹系统：预分配 128 颗 Bullet，所有武器共享这个池。发射时 Get，命中/超出距离时 Return。整局游戏零 GC 分配。

# Unity 官方 Particle System

每个粒子内部就是对象池实现。持续发射的粒子系统如果每帧 new Particle()，一秒钟就能吃掉几 MB 堆内存。

# 注意事项

# 不要池化什么

  • 生命周期贯穿整个游戏的对象（主角、永久 UI）→ 池化没意义
  • 尺寸差异极大的对象 → 池子按最大尺寸分配，浪费内存；按需创建更划算
  • 初始化开销极大的对象 → 比如加载场景、编译 Shader，预热成本本身就大

# 调试技巧

给池化对象加一个 _poolId 标记：

#if DEBUG
    bullet._poolId = GetHashCode();
#endif

当诡异 bug 出现（比如一个"已回收"的子弹还在屏幕上），能追踪到哪个池、哪个时间点。

# 与其他模式的配合

  • 对象池 + ECS：ECS 的 Component 数组本身就是一个"池"，删除 Entity 时 Component 不会释放内存，只是标记为可用槽位
  • 对象池 + Command Pattern：Command 对象也可以池化，每个输入事件复用同一个 Command 实例

一句话总结： 堆分配是游戏性能的隐形杀手，对象池不是优化技巧，而是必备基础设施。在射击游戏里，没有对象池就等着 GC 把帧率拉垮。
