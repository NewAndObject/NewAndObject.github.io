# 懒加载：用到才加载

class TextureManager {
    Dictionary<string, Texture2D> cache = new();

    public Texture2D Get(string path) {
        if (!cache.ContainsKey(path)) {
            cache[path] = Resources.Load<Texture2D>(path);  // 用到才加载
        }
        return cache[path];
    }
}

# 引用计数：自动管理生命周期

class RefCountedTexture {
    public Texture2D texture;
    public int refCount;

    public void Retain() { refCount++; }
    public void Release() {
        refCount--;
        if (refCount <= 0) {
            Resources.UnloadAsset(texture);  // 无人引用，卸载
            texture = null;
        }
    }
}

# 完整实现

class ResourceManager {
    private Dictionary<string, RefCountedTexture> textures = new();

    public Texture2D Load(string path) {
        if (!textures.ContainsKey(path)) {
            RefCountedTexture rt = new();
            rt.texture = Resources.Load<Texture2D>(path);
            rt.refCount = 0;
            textures[path] = rt;
        }
        textures[path].Retain();
        return textures[path].texture;
    }

    public void Unload(string path) {
        if (textures.TryGetValue(path, out var rt)) {
            rt.Release();
            if (rt.refCount <= 0) {
                textures.Remove(path);
            }
        }
    }
}

# 在 Godot 中的实现

class_name ResourceCache

static var _refs: Dictionary = {}

static func load(path: String) -> Resource:
    if not _refs.has(path):
        _refs[path] = {
            "refs": 0,
            "resource": load(path)
        }
    _refs[path].refs += 1
    return _refs[path].resource

static func unload(path: String):
    if _refs.has(path):
        _refs[path].refs -= 1
        if _refs[path].refs <= 0:
            _refs[path].resource.free()
            _refs.erase(path)

static func get_refcount(path: String) -> int:
    return _refs.get(path, {}).get("refs", 0)

# 弱引用变体：避免循环引用

有时候 A 资源引用了 B，B 引用了 A，双方 refCount 永远不会归零。用 WeakReference 打破循环：

class WeakResourceRef {
    WeakReference<Texture2D> weakRef;

    public Texture2D Get() {
        if (weakRef.TryGetTarget(out var tex))
            return tex;
        return null;  // 资源已被卸载，调用方自行处理
    }
}

# 引用计数 vs GC

引用计数 GC（垃圾回收）
资源释放时机 立即（refCount = 0 时） 不确定（GC 触发时）
性能开销 每次 Retain / Release 暂停所有线程扫描
循环引用 需要 WeakRef 处理 自动处理
确定性 ✅ 你知道什么时候释放 ❌ 你不知道

游戏资源管理用引用计数，游戏逻辑对象用 GC —— 这是大多数商业引擎的选择。

# 更高级的变体

# 1. 分级缓存（LOD Cache）

不同"热度"的资源使用不同的生命周期策略：

enum CacheLevel {
    Permanent,     // 永远不卸载（UI、核心角色）
    SceneLevel,    // 场景切换时卸载
    LRU,           // 最近最少使用，内存不足时淘汰
    Pooled         // 随对象池自动管理
}

# 2. 预算管理（Budget Manager）

给每种资源类型设内存上限：

class TextureBudget {
    const long MAX_MEMORY = 512 * 1024 * 1024;  // 512MB
    long currentMemory = 0;
    List<RefCountedTexture> loaded = new();

    Texture2D Load(string path) {
        var tex = LoadInternal(path);
        currentMemory += tex.width * tex.height * 4;

        // 超预算？卸载 LRU 纹理
        while (currentMemory > MAX_MEMORY) {
            var lru = loaded.OrderBy(t => t.lastAccessTime).First();
            if (lru.refCount > 0) break;  // 正在用的不能卸
            UnloadInternal(lru);
            currentMemory -= lru.size;
        }
        return tex;
    }
}

# 3. 预加载（Preload / 预热）

在加载界面提前加载下一场景需要的资源：

class LevelLoader {
    async void PreloadLevel(string levelName) {
        var resources = GetRequiredResources(levelName);
        var tasks = resources.Select(r => ResourceManager.LoadAsync(r));
        await Task.WhenAll(tasks);

        // 预加载完成，切换场景
        SceneManager.LoadScene(levelName);
    }
}

# 实际项目建议

  1. 小游戏（< 50MB 资源） — 全部加载，不需要引用计数，启动快更重要
  2. 中型游戏 — 懒加载 + 场景切换时全卸载
  3. 大型游戏（开放世界） — 懒加载 + 引用计数 + 预算管理 + Streaming

# 典型反模式

❌ 每帧都 Load/Unload 同一个资源
   → 在 Update 里反复加载同一个纹理，性能灾难

❌ 用完之后不 Unload
   → 内存泄漏，游戏越来越卡

❌ 在热循环里做同步加载
   → 每帧 Load 触发磁盘 I/O，帧率暴跌

✅ 正确做法：
   Load → 缓存 → Retain → Release → refCount=0 时自动 Unload
