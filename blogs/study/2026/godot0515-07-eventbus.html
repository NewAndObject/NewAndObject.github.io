# 基本实现

class EventBus {
    private Dictionary<Type, List<Delegate>> handlers = new();

    public void Subscribe<T>(Action<T> handler) where T : IEvent {
        var type = typeof(T);
        if (!handlers.ContainsKey(type))
            handlers[type] = new List<Delegate>();
        handlers[type].Add(handler);
    }

    public void Unsubscribe<T>(Action<T> handler) where T : IEvent {
        var type = typeof(T);
        if (handlers.TryGetValue(type, out var list))
            list.Remove(handler);
    }

    public void Fire<T>(T eventData) where T : IEvent {
        var type = typeof(T);
        if (handlers.TryGetValue(type, out var list)) {
            foreach (Action<T> handler in list) {
                handler(eventData);
            }
        }
    }
}

# 订阅与发布

// 初始化时订阅
class AchievementSystem {
    void Awake() {
        EventBus.Instance.Subscribe<PlayerDiedEvent>(OnPlayerDied);
        EventBus.Instance.Subscribe<ItemCollectedEvent>(OnItemCollected);
    }

    void OnPlayerDied(PlayerDiedEvent e) {
        if (e.killer is Boss) Unlock("直面BOSS");
        if (e.deathPosition.y < -1000) Unlock("坠崖者");
    }

    void OnItemCollected(ItemCollectedEvent e) {
        if (e.itemId == "diamond" && e.count >= 10) Unlock("钻石恒久远");
    }
}

// 触发时发布
class Player {
    void Die(Enemy killer) {
        EventBus.Instance.Fire(new PlayerDiedEvent {
            player = this,
            killer = killer,
            deathPosition = transform.position
        });
    }
}

# 在 Godot 中的实现

# EventBus.gd (Autoload)
extends Node

var _handlers: Dictionary = {}

func subscribe(event_name: String, target: Object, method: String):
    if not _handlers.has(event_name):
        _handlers[event_name] = []
    _handlers[event_name].append({ "target": target, "method": method })

func unsubscribe(event_name: String, target: Object, method: String):
    if _handlers.has(event_name):
        _handlers[event_name] = _handlers[event_name].filter(
            func(h): return h.target != target or h.method != method
        )

func fire(event_name: String, data: Dictionary = {}):
    if _handlers.has(event_name):
        for h in _handlers[event_name]:
            if is_instance_valid(h.target):
                h.target.call(h.method, data)

# 常见应用场景

# 1. 成就系统

最经典的事件总线用例：

// 几十个成就 = 几十个事件订阅，全是独立的
EventBus.Subscribe<EnemyKilledEvent>(e => {
    if (e.count >= 100) Unlock("百人斩");
});
EventBus.Subscribe<GoldCollectedEvent>(e => {
    if (e.total >= 10000) Unlock("万元户");
});
EventBus.Subscribe<LevelUpEvent>(e => {
    if (e.newLevel >= 50) Unlock("半百");

# 事件总线 vs 命令模式

事件总线 Command
语义 "某件事发生了" "请执行这个操作"
谁触发 发布者广播 调用者直接请求
响应者 0 ~ N 个 1 个
可取消？ 通常不行（已发生） 可以（还没执行）
典型用途 成就、UI、音效、分析 RTS 指令、撤销、回放

实际项目两者同时使用：

Input → Command（直接执行） → Event（通知旁观系统）
                                    ↓
                          成就系统 / UI / 音效 / Mod

# 注意事项

# 1. 不要滥用

事件总线不是万能胶。两个系统有明确依赖关系时，直接调用比事件更清晰。

// ❌ 过度使用事件总线
EventBus.Fire(new SaveGameRequest());  // 为什么不直接 SaveManager.Save()？

// ✅ 用了事件总线的后果：你不知道谁响应了 SaveGameRequest，代码理解困难

# 2. 内存泄漏

订阅者忘记取消订阅 = 发布者持有对订阅者的引用 → 订阅者无法被 GC：

class MyBehaviour : MonoBehaviour {
    void OnEnable() {
        EventBus.Subscribe<DamageEvent>(OnDamage);
    }
    void OnDisable() {
        EventBus.Unsubscribe<DamageEvent>(OnDamage);  // 必须取消！
    }
}

# 3. 调试困难

事件多了之后，"这个事件谁发的？谁在响应？" 难以追踪。

解决方案：加调试模式，打印事件流：

public void Fire<T>(T data) {
#if DEBUG
    Debug.Log($"[EventBus] {typeof(T).Name}");
#endif
    // ... 正常派发
}

# 实际项目建议

项目规模 是否用事件总线 原因
小项目（1-2人） 可不用 系统少，直接依赖更简单
中型项目 推荐 成就/UI/Mod 场景下收益明显
大型项目 核心设施 系统多到互相不知道存在

一句话总结： 事件总线让"发声"和"听声"解耦 —— 不想让对方知道自己存在的系统之间，事件总线是唯一的通信方式。
