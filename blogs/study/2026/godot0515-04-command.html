# Command Pattern：把操作变成对象

# 问题：紧耦合的输入处理

最直接的写法：

void Update() {
    if (Input.GetKeyDown(KeyCode.W))  player.MoveForward();
    if (Input.GetKeyDown(KeyCode.S))  player.MoveBack();
    if (Input.GetKeyDown(KeyCode.A))  player.MoveLeft();
    if (Input.GetKeyDown(KeyCode.D))  player.MoveRight();
    if (Input.GetKeyDown(KeyCode.Space)) player.Jump();
}

问题：

  • 输入和逻辑绑死了 — 换手柄/触屏要改 Update()
  • 不能撤销 — 操作完就没了
  • 不能回放 — 没办法记录和回播操作序列
  • 不能延迟执行 — "选中单位 → 移动指令" 这种两步操作需要额外状态

# Command Pattern 的思想

把"一个操作"封装成一个对象：

Input → Command（对象）→ Execute() → 改变游戏状态
1

每个 Command 对象包含执行这个操作需要的全部信息。

# 基本实现

# 1. 定义 Command 接口

interface ICommand {
    void Execute();
    void Undo();
}

# 2. 具体命令

class MoveCommand : ICommand {
    Player player;
    Vector3 from, to;

    void Execute() {
        from = player.position;
        player.position = to;
    }

    void Undo() {
        player.position = from;
    }
}

class JumpCommand : ICommand {
    Player player;

    void Execute() {
        player.Jump();
    }

    void Undo() {
        // 跳跃不能撤销，留空或什么都不做
    }
}

class AttackCommand : ICommand {
    Player player;
    float timestamp;

    void Execute() {
        player.Attack();
    }

    void Undo() { }  // 攻击一般不能撤销
}

# 3. 记录与执行

class CommandManager {
    Stack<ICommand> history = new();
    Stack<ICommand> redoStack = new();

    void ExecuteCommand(ICommand cmd) {
        cmd.Execute();
        history.Push(cmd);
        redoStack.Clear();  // 新操作后，redo 历史清空
    }

    void Undo() {
        if (history.Count == 0) return;
        var cmd = history.Pop();
        cmd.Undo();
        redoStack.Push(cmd);
    }

    void Redo() {
        if (redoStack.Count == 0) return;
        var cmd = redoStack.Pop();
        cmd.Execute();
        history.Push(cmd);
    }
}

# 在游戏中的应用

# 1. RTS 即时战略游戏

这就是 RTS 的核心架构：

// 玩家操作
Command selectCmd = new SelectCommand(unit, screenRect);
Command moveCmd = new MoveCommand(selectedUnits, targetPosition);
Command buildCmd = new BuildCommand(builder, buildingType, location);

cmdManager.ExecuteCommand(moveCmd);

// 网络同步：序列化 Command 发给服务器
byte[] data = moveCmd.Serialize();
network.Send(data);

// 回放：把 Command 列表存成文件
replayFile.Write(moveCmd.Serialize());

# 2. 角色技能系统

class SkillCommand : ICommand {
    Unit caster;
    Unit target;
    Skill skill;
    int hpBefore;

    void Execute() {
        hpBefore = target.currentHP;
        skill.Apply(caster, target);
    }

    void Undo() {
        target.currentHP = hpBefore;
        target.RemoveStatusEffects(skill.id);
    }
}

# 3. 关卡编辑器

任何一个关卡编辑器（包括游戏里的建筑模式）都应该用 Command Pattern：

class PlaceObjectCommand : ICommand {
    Vector3 position;
    string objectId;
    GameObject obj;

    void Execute() {
        obj = Instantiate(prefab, position);
        SceneManager.current.objects.Add(obj);
    }

    void Undo() {
        Destroy(obj);
        SceneManager.current.objects.Remove(obj);
    }
}

class DeleteObjectCommand : ICommand {
    GameObject obj;
    Vector3 originalPos;

    void Execute() {
        originalPos = obj.transform.position;
        obj.SetActive(false);
    }

    void Undo() {
        obj.SetActive(true);
        obj.transform.position = originalPos;
    }
}

        player.MoveTo(new Vector3(targetX, targetY));

# 5. 输入缓冲 / 延迟输入

格斗游戏里玩家输入和角色动作之间有微小延迟（输入缓冲）：

class InputBuffer {
    Queue<ICommand> buffer = new();

    void AddInput(ICommand cmd) {
        buffer.Enqueue(cmd);
        if (buffer.Count > 10) buffer.Dequeue();  // 最多保留10帧的输入
    }

    // 每帧取出缓冲区里的输入去执行
    void FlushToCharacter(Character c) {
        while (buffer.Count > 0) {
            buffer.Dequeue().Execute();
        }
    }
}

# Command 模式的变体

# 延迟命令

不立即执行，等到条件满足再执行：

class DelayedCommand : ICommand {
    ICommand inner;
    float delay;

    void Execute() {
        if (delay > 0) {
            delay -= Time.deltaTime;
            return;  // 还没到时间，不执行
        }
        inner.Execute();
    }
}

# 组合命令（Macro）

class MacroCommand : ICommand {
    List<ICommand> commands;

    void Execute() {
        foreach (var cmd in commands) cmd.Execute();
    }

    void Undo() {
        for (int i = commands.Count - 1; i >= 0; i--)
            commands[i].Undo();
    }
}

// 一键"建造兵营" = 选择工人 + 走到位置 + 开始建造
var buildBarracks = new MacroCommand(new List<ICommand> {
    new SelectCommand(worker),
    new MoveCommand(worker, location),
    new BuildCommand("barracks", location)
});

# 实际项目建议

  1. RTS / MOBA / 战棋 — Command Pattern 就是核心架构，输入输出全是 Command
  2. 格斗游戏 — 输入缓冲 + 回滚网络必须用 Command
  3. 关卡编辑器 / 建造模式 — 不用 Command 就别想支持 Undo
  4. 单机 RPG — 不是必需品，但技能系统用 Command 会清晰很多

# 对比：Command vs Event

Command Event
目的 封装一个待执行的操作 广播一个已经发生的事
有 Undo？ 有（设计时就考虑） 没有
可序列化？ 是 不一定
可排队？ 是 不是
例子 MoveCommand OnEnemyDied

一句话总结： Command 模式把"正在做的事"和"怎么处理输入"解耦，是撤销/回放/网络同步/输入缓冲这四个核心系统的共同基石。
