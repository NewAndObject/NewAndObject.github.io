# 数据与表现分离：写一次逻辑，换任意皮肤

# 问题：逻辑和渲染混在一起

class Monster {
    float hp = 100;

    void TakeDamage(float damage) {
        hp -= damage;
        // 混杂表现代码
        PlayAnimation("hurt");           // 播放受伤动画
        ShowDamageNumber(damage);        // 显示伤害数字
        CameraShake();                   // 屏幕震动
        PlaySound("hit");               // 播放音效
        if (hp <= 0) {
            PlayAnimation("die");
            DropLoot();
            SpawnDeathParticles();
        }
    }
}

这段代码的问题：

  • 没办法 Headless 运行（服务器怎么跑？没有 Camera、没有动画系统）
  • 测试困难 — 写单元测试时要 Mock 动画系统、音效系统
  • 改表现必须改逻辑代码 — 想把受伤效果从"闪红"改成"变灰" → 改 Monster 类
  • 多个表现端没法复用 — 同一套逻辑没法同时支持 PC 端和手机端的差异表现

# 分离的核心

数据 / 逻辑层（Model）       表现层（View / Presentation）
───────────────────         ─────────────────────
纯数据，无任何依赖渲染       只管"怎么展示"，不管"为什么"
可 Headless 运行             可以换，可以没有
可单元测试                   属于"终端"设备

# 基本架构

// ===== 数据层：纯逻辑 =====
class CharacterData {
    public float hp;
    public float maxHP;
    public Vector3 position;
    public bool isDead => hp <= 0;

    public event Action<float> OnDamageTaken;
    public event Action OnDied;

    public void TakeDamage(float damage) {
        hp = Mathf.Max(0, hp - damage);
        OnDamageTaken?.Invoke(damage);
        if (isDead) OnDied?.Invoke();
    }
}

// ===== 表现层：只管展示 =====
class CharacterView : MonoBehaviour {
    public CharacterData data;
    public Animator animator;
    public DamageNumber damageNumberPrefab;
    public ParticleSystem deathParticles;

    void Start() {
        data.OnDamageTaken += PlayHurtEffect;
        data.OnDied += PlayDeathEffect;
    }

    void PlayHurtEffect(float damage) {
        animator.SetTrigger("hurt");
        Instantiate(damageNumberPrefab).Show(damage);
        CameraShake.Instance.Shake(0.1f);
    }

    void PlayDeathEffect() {
        animator.SetTrigger("die");
        Instantiate(deathParticles, transform.position);
    }
}

// ===== 使用 =====
var data = new CharacterData { hp = 100, maxHP = 100 };
var view = Instantiate(characterPrefab).GetComponent<CharacterView>();
view.data = data;

// 逻辑层处理伤害
data.TakeDamage(30);  // ← view 自动响应

# 在 Godot 中的实现

# ===== 数据层 =====
class_name CharacterData
extends Resource

@export var hp: float = 100.0
@export var max_hp: float = 100.0
@export var speed: float = 200.0

signal damage_taken(amount: float)
signal died()

func take_damage(amount: float) -> void:
    hp = max(0, hp - amount)
    damage_taken.emit(amount)
    if hp <= 0:
        died.emit()

# ===== 表现层 =====
extends CharacterBody2D

@export var data: CharacterData
@onready var anim: AnimationPlayer = $AnimationPlayer

func _ready() -> void:
    data.damage_taken.connect(_on_damage_taken)
    data.died.connect(_on_died)

func _on_damage_taken(amount: float) -> void:
    anim.play("hurt")
    $DamageNumber.show(amount)
    $HitSound.play()

func _on_died() -> void:
    anim.play("die")
    await anim.animation_finished
    queue_free()

# 分离带来的关键好处

# 1. Headless 服务器

多人游戏中，服务器只关心逻辑层：

// 服务器（无渲染）
class GameServer {
    CharacterData playerData = new();

    void OnClientDamageRequest(float amount) {
        playerData.TakeDamage(amount);  // 纯逻辑，任何渲染代码都不该存在
        SyncToClients(playerData);
    }
}

// 客户端
class GameClient {
    void OnSync(CharacterData serverData) {
        localPlayerData.hp = serverData.hp;  // 接受服务器权威数据
    }
}

# 4. 支持回放

回放系统只需要记录数据层的变更，不需要记录每一帧的渲染状态：

记录：
  Frame 100: TakeDamage(30) → HP: 100→70
  Frame 150: TakeDamage(50) → HP: 70→20
  Frame 200: Heal(40)       → HP: 20→60

回放时重放这些数据变更 → 表现层自动"演"出动画

# 预测与回滚

格斗游戏和 FPS 中，客户端在自己本地先"预测"执行操作来消除延迟感：

class PredictionSystem {
    CharacterData localData;
    Stack<CharacterData> history = new();  // 保存历史帧用于回滚

    void ClientMove(Vector3 dir) {
        history.Push(localData.Clone());   // 备份
        localData.position += dir * speed * Time.deltaTime;  // 预测
    }

    void OnServerSync(CharacterData authoritative) {
        if (!localData.Equals(authoritative)) {
            // 预测错了，回滚到上一个已知正确状态
            localData = authoritative;
            // 重新应用本地输入
            foreach (var input in pendingInputs) {
                localData.ApplyInput(input);
            }
        }
    }
}

# 实际项目建议

# 什么时候必须分离

  • 多人游戏 — 服务器不需要渲染
  • 需要单元测试 — 测试逻辑不能依赖渲染管线
  • 多平台发布 — PC + 主机 + 手机共享逻辑
  • 回放系统 — 需要纯数据重放

# 什么时候不必过度分离

  • 纯单机小型游戏 — 分离增加接口和事件，代码量翻倍
  • 原型 / Game Jam — 快速出东西比架构更重要

# 分离的度

// 松耦合（推荐）：数据层通过事件通知表现层
data.TakeDamage(30);  // 只改数据
// 表现层通过事件自动响应

// 紧耦合（不推荐）：逻辑层直接调用表现层
TakeDamage(30);
PlayHurtAnimation();  // 逻辑不应该关心这个

一句话总结： 数据层回答"发生了什么"，表现层回答"怎么展示"。改前者不影响后者，换后者不碰前者。这是游戏架构的第一原则。
